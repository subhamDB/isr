# ============================================================
# AUTOENCODER ANOMALY DETECTION USING MNIST DATASET
# ============================================================

# 1. Import Required Libraries
import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt

from sklearn.metrics import accuracy_score, confusion_matrix, classification_report

# 2. Load Dataset
(x_train, y_train), (x_test, y_test) = tf.keras.datasets.mnist.load_data()

print("Training Data Shape:", x_train.shape)
print("Testing Data Shape:", x_test.shape)

# 3. Normalize Data
x_train = x_train.astype("float32") / 255.0
x_test = x_test.astype("float32") / 255.0

# Select digit 0 as normal data
normal_data = x_train[y_train == 0]

# Flatten images
normal_data = normal_data.reshape(-1, 784)
x_test = x_test.reshape(-1, 784)

print("Normal Training Data:", normal_data.shape)

# 4. Build Encoder
encoder = tf.keras.Sequential([
    tf.keras.layers.Input(shape=(784,)),
    tf.keras.layers.Dense(128, activation='relu'),
    tf.keras.layers.Dense(64, activation='relu')
], name="Encoder")

# 5. Build Decoder
decoder = tf.keras.Sequential([
    tf.keras.layers.Input(shape=(64,)),
    tf.keras.layers.Dense(128, activation='relu'),
    tf.keras.layers.Dense(784, activation='sigmoid')
], name="Decoder")

# 6. Build Autoencoder
model = tf.keras.Sequential([
    encoder,
    decoder
], name="Autoencoder")

model.summary()

# 7. Compile Model
model.compile(
    optimizer='adam',
    loss='mse',
    metrics=['mae']
)

# 8. Train Model
history = model.fit(
    normal_data,
    normal_data,
    epochs=5,
    batch_size=128,
    validation_split=0.1,
    shuffle=True
)

# 9. Plot Training Loss
plt.figure(figsize=(8, 4))
plt.plot(history.history['loss'], label='Training Loss')
plt.plot(history.history['val_loss'], label='Validation Loss')
plt.title("Autoencoder Training Loss")
plt.xlabel("Epochs")
plt.ylabel("Loss")
plt.legend()
plt.show()

# 10. Reconstruct Test Images
predictions = model.predict(x_test, verbose=0)

# Display Original and Reconstructed Images
plt.figure(figsize=(12, 5))

for i in range(5):
    plt.subplot(2, 5, i + 1)
    plt.imshow(x_test[i].reshape(28, 28), cmap='gray')
    plt.title("Original")
    plt.axis('off')

    plt.subplot(2, 5, i + 6)
    plt.imshow(predictions[i].reshape(28, 28), cmap='gray')
    plt.title("Reconstructed")
    plt.axis('off')

plt.tight_layout()
plt.show()

# 11. Calculate Reconstruction Error
error = np.mean((x_test - predictions) ** 2, axis=1)

# Calculate Normal Training Error
normal_predictions = model.predict(normal_data, verbose=0)

normal_error = np.mean(
    (normal_data - normal_predictions) ** 2,
    axis=1
)

# Calculate Anomaly Threshold
threshold = np.percentile(normal_error, 95)

print("Anomaly Threshold:", threshold)

# 12. Detect Anomalies
# 0 = Normal, 1 = Anomaly

predicted_labels = (error > threshold).astype(int)

print("\nAnomaly Detection Results:")

for i in range(20):
    if predicted_labels[i] == 1:
        print(i, "-> Anomaly")
    else:
        print(i, "-> Normal")

# 13. Calculate Accuracy
actual_labels = (y_test != 0).astype(int)

accuracy = accuracy_score(actual_labels, predicted_labels)

print("\nAccuracy:", round(accuracy * 100, 2), "%")

# 14. Confusion Matrix and Classification Report
print("\nConfusion Matrix:")
print(confusion_matrix(actual_labels, predicted_labels))

print("\nClassification Report:")
print(classification_report(
    actual_labels,
    predicted_labels,
    target_names=['Normal', 'Anomaly']
))

# 15. Plot Anomaly Detection Graph
plt.figure(figsize=(10, 6))

plt.hist(
    error[actual_labels == 0],
    bins=50,
    alpha=0.7,
    label='Normal'
)

plt.hist(
    error[actual_labels == 1],
    bins=50,
    alpha=0.7,
    label='Anomaly'
)

plt.axvline(
    threshold,
    color='black',
    linestyle='--',
    label='Threshold'
)

plt.xlabel("Reconstruction Error")
plt.ylabel("Number of Samples")
plt.title("Autoencoder Anomaly Detection")
plt.legend()
plt.grid(True)
plt.show()

# 16. Final Results
print("\n====================================")
print(" AUTOENCODER ANOMALY DETECTION")
print("====================================")
print("Training Samples:", len(normal_data))
print("Testing Samples:", len(x_test))
print("Anomaly Threshold:", round(threshold, 6))
print("Accuracy:", round(accuracy * 100, 2), "%")
print("Predicted Normal:", np.sum(predicted_labels == 0))
print("Predicted Anomalies:", np.sum(predicted_labels == 1))
print("====================================")
