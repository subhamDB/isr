# 1. Import Libraries
import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt

# 2. Load Dataset
(x_train, y_train), (x_test, y_test) = tf.keras.datasets.mnist.load_data()

# 3. Normalize Data
x_train = x_train / 255.0
x_test = x_test / 255.0

# Use digit 0 as normal data
normal_data = x_train[y_train == 0]

# Flatten Images
normal_data = normal_data.reshape(-1, 784)
x_test = x_test.reshape(-1, 784)

# 4. Build Autoencoder
model = tf.keras.Sequential([
    tf.keras.layers.Input(shape=(784,)),
    tf.keras.layers.Dense(128, activation='relu'),
    tf.keras.layers.Dense(64, activation='relu'),
    tf.keras.layers.Dense(128, activation='relu'),
    tf.keras.layers.Dense(784, activation='sigmoid')
])

# 5. Compile Model
model.compile(
    optimizer='adam',
    loss='mse'
)

# 6. Train Model
model.fit(
    normal_data,
    normal_data,
    epochs=5,
    batch_size=128
)

# 7. Reconstruct Test Data
predictions = model.predict(x_test)

# Display Original and Reconstructed Images
plt.figure(figsize=(12, 5))

for i in range(5):
    plt.subplot(2, 5, i + 1)
    plt.imshow(x_test[i].reshape(28, 28), cmap='gray')
    plt.title("Original")
    plt.axis('off')

    plt.subplot(2, 5, i + 6)
    plt.imshow(predictions[i].reshape(28, 28), cmap='gray')
    plt.title("Reconstructed")
    plt.axis('off')

plt.tight_layout()
plt.show()

# 8. Calculate Reconstruction Error
error = np.mean((x_test - predictions) ** 2, axis=1)

normal_predictions = model.predict(normal_data)

threshold = np.percentile(
    np.mean((normal_data - normal_predictions) ** 2, axis=1),
    95
)

# 9. Detect Anomaly
results = (error > threshold).astype(int)

print("Anomaly Detection Results:")

for i in range(20):
    if results[i] == 1:
        print(i, "-> Anomaly")
    else:
        print(i, "-> Normal")

# 10. Plot Anomaly Detection Graph
plt.figure(figsize=(10, 6))

plt.hist(error[results == 0], bins=50,
         alpha=0.7, label='Normal (0)')

plt.hist(error[results == 1], bins=50,
         alpha=0.7, label='Anomaly (1)')

plt.axvline(threshold, color='black',
            linestyle='--', label='Threshold')

plt.xlabel("Reconstruction Error")
plt.ylabel("Number of Samples")
plt.title("Autoencoder Anomaly Detection")

plt.legend()
plt.grid(True)
plt.tight_layout()
plt.show()
